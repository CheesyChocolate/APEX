\section{Web Platform}

\begin{frame}{APEX Web Platform}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Virtual Screening Mode}
        Users supply their own compound library for screening:
        \begin{itemize}
          \item CSV upload — SMILES column auto-detected
          \item Manual SMILES entry (single or batch)
          \item Demo mode: evaluates on ChEMBL test-split compounds
        \end{itemize}
        \vspace{0.15cm}
        Train on known ChEMBL actives, predict on a novel
        library — the scientifically correct use case for
        identifying new scaffolds.
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Results Interface}
        \begin{itemize}
          \item QSAR table ranked by activity probability
          \item Lipinski descriptors per compound
          \item Docking hits ranked by $\Delta G_{\text{bind}}$
          \item Affinity heat colouring for rapid triage
          \item Side-by-side compound comparison
        \end{itemize}
      \end{block}
      \vspace{0.2cm}
      \begin{exampleblock}{Any Target, On Demand}
        Search by name or ChEMBL ID — the full pipeline
        runs automatically with no manual configuration.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}
