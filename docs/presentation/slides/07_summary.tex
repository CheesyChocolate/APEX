\section{Summary \& Future Work}

\begin{frame}{Summary \& Future Work}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{What APEX Delivers}
        \begin{itemize}
          \item Fully automated, target-agnostic screening pipeline
          \item ChEMBL bioactivity ingestion with local caching
          \item QSAR model (RF, AUC $>$ 0.94 on CDK2)
          \item AutoDock Vina docking for top-N candidates
          \item REST API + interactive web frontend
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{exampleblock}{Planned Extensions}
        \begin{itemize}
          \item Larger virtual compound libraries (Enamine REAL)
          \item Ensemble QSAR models (XGBoost, GNN)
          \item ADMET property filtering pre-docking
          \item 3D binding pose visualisation in-browser
          \item Multi-target comparative screening
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}

  \vspace{0.5cm}
  \begin{center}
    \large\textbf{APEX}: an open, extensible platform that turns a target name
    into a ranked hit list — fully automatically.
  \end{center}
