\section{Summary \& Future Work}

\begin{frame}{Summary \& Future Work}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{What APEX Delivers}
        \begin{itemize}
          \item Target-agnostic pipeline: ChEMBL $\rightarrow$ QSAR $\rightarrow$ docking
          \item RF classifier on $\mathbb{R}^{1030}$ (Morgan + 6 descriptors)
          \item Lipinski filter + max-min diversity pre-selection
          \item AutoDock Vina with adaptive binding-box estimation
          \item Virtual screening: user-supplied compound libraries
          \item REST API + interactive web frontend, no build step
        \end{itemize}
      \end{block}
      \vspace{0.2cm}
      \begin{block}{CDK2 Validation}
        CV AUC\,=\,0.943 \quad Test F1\,=\,0.901\\
        Top hit: $-9.5$\,kcal/mol \quad Pipeline: $<$10\,min
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{exampleblock}{Planned Extensions}

          \item Automated binding-site detection (fpocket / P2Rank)
          \item Multi-target comparative screening
        \end{itemize}
      \end{exampleblock}
    \end{column}
