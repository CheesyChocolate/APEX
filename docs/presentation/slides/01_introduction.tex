\section{Introduction}

\begin{frame}{The Drug Discovery Challenge}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{Scale of the Problem}
        \begin{itemize}
          \item 12--15 years from target to approval
          \item \$2.6 billion average per approved drug
          \item $\sim$1 in 10{,}000 screened compounds reaches trials
        \end{itemize}
      \end{block}
      \vspace{0.4cm}
      \begin{exampleblock}{Computational Acceleration}
        Early-stage virtual screening reduces the wet-lab search
        space from millions to tens of candidates before any
        synthesis is attempted.
      \end{exampleblock}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{alertblock}{Bottleneck}
        Manual, target-specific pipelines require expert setup for
        every new biological target — not scalable.
      \end{alertblock}
      \vspace{0.4cm}
      \begin{block}{Our Approach}
        \textbf{APEX} automates the full pipeline:
        data ingestion $\rightarrow$ QSAR $\rightarrow$ docking
        for \emph{any} ChEMBL target, on demand.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
